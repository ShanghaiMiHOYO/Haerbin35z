function initUploader() {
  const fileInput = document.getElementById('fileInput');
  const selectBtn = document.getElementById('selectBtn');
  const dropZone = document.getElementById('dropZone');
  const previewList = document.getElementById('previewList');
  const uploadBtn = document.getElementById('uploadBtn');
  const authorInput = document.getElementById('authorInput');
  let pendingFiles = [];

  selectBtn.onclick = () => fileInput.click();
  fileInput.onchange = e => handleFiles(e.target.files);

  ['dragenter','dragover'].forEach(evt => {
    dropZone.addEventListener(evt,e=>{e.preventDefault(); dropZone.classList.add('drop-active')});
  });
  ['dragleave','drop'].forEach(evt => {
    dropZone.addEventListener(evt,e=>{e.preventDefault(); dropZone.classList.remove('drop-active')});
  });
  dropZone.ondrop = e => handleFiles(e.dataTransfer.files);

  function handleFiles(files) {
    Array.from(files).forEach(file => {
      const id = getId();
      const entry = {id, file, name:file.name, type:file.type, size:file.size};
      pendingFiles.push(entry);
      renderPreview(entry);
    });
  }

  function renderPreview(entry) {
    const card = document.createElement('div');
    card.className = 'bg-white/5 rounded p-3 flex gap-3';
    const media = document.createElement('div');
    media.className = 'w-36 h-24 bg-black/20 rounded overflow-hidden';
    if (entry.type.startsWith('image/')) {
      const img = document.createElement('img');
      img.className = 'w-full h-full object-cover';
      img.src = URL.createObjectURL(entry.file);
      media.appendChild(img);
    } else if (entry.type.startsWith('video/')) {
      const v = document.createElement('video');
      v.className = 'w-full h-full object-cover';
      v.src = URL.createObjectURL(entry.file);
      v.controls = true;
      media.appendChild(v);
    }
    const info = document.createElement('div');
    info.className = 'flex-1';
    info.innerHTML = `<div class="font-medium">${entry.name}</div>
    <div class="text-sm text-blue-200">${formatBytes(entry.size)}</div>`;
    card.appendChild(media);
    card.appendChild(info);
    previewList.appendChild(card);
  }

  // 提交审核（用户）
  uploadBtn.onclick = () => {
    const author = authorInput.value.trim();
    if (!author) return alert('请填写作者姓名');
    if (pendingFiles.length === 0) return alert('请选择文件');
    pendingFiles.forEach(entry => {
      const blobUrl = URL.createObjectURL(entry.file);
      const item = {
        id: entry.id, name: entry.name, type: entry.type,
        author, blobUrl, time: new Date().toLocaleString()
      };

    });
    pendingFiles = [];

    renderAudit();
  };

    if (pwd !== ADMIN_PASSWORD) return alert('密码错误');
    document.getElementById('auditPanel').classList.remove('hidden');
    renderAudit();
  };

function renderAudit() {
  const grid = document.getElementById('auditGrid');
  grid.innerHTML = '';

    info.textContent = `${item.name}｜${item.author}`;
    card.appendChild(info);

    const tag = document.createElement('div');
    tag.className = 'status-tag status-pending';

    card.appendChild(tag);

    box.className = 'p-4 flex gap-3';
    const ok = document.createElement('button');
    ok.className = 'audit-pass flex-1';

    no.className = 'audit-reject flex-1';
    no.textContent = '❌ 拒绝';

    box.appendChild(ok);
    box.appendChild(no);
    card.appendChild(box);

    card.onclick = (e) => {
      if (e.target.tagName === 'BUTTON') return;
      openMediaModal(item.blobUrl, item.type.startsWith('video/')?'video':'image', item.name);
    };
    grid.appendChild(card);
  });
}

// 审核操作
function passItem(id) {
  const pending = get(STORAGE_PENDING);
  const item = pending.find(i => i.id === id);
  if (!item) return;
  const passed = get(STORAGE_PASSED);
  passed.unshift(item);
  set(STORAGE_PASSED, passed);
  const newPending = pending.filter(i => i.id !== id);
  set(STORAGE_PENDING, newPending);
  renderAudit();
  renderPassed();
}
function rejectItem(id) {
  if (!confirm('确定拒绝并删除？')) return;
  const pending = get(STORAGE_PENDING).filter(i => i.id !== id);
  set(STORAGE_PENDING, pending);
  renderAudit();
}

// 媒体模态
function openMediaModal(src, type, name) {
  if (!window.__modal) createModal();
  window.openMediaModal(src, type, name);
}
function createModal() {
  const m = document.createElement('div');
  m.className = 'fixed inset-0 z-50 flex items-center justify-center bg-black/75';
  m.innerHTML = `<div class="bg-neutral-900 rounded relative p-2 max-w-[90%] max-h-[90%]">
    <button id="mclose" class="absolute right-2 top-2 bg-red-600 px-2 rounded text-white">✕</button>
    <div id="mbody" class="flex items-center justify-center"></div>
    <div id="mtext" class="text-white p-2 text-center"></div>
  </div>`;
  document.body.appendChild(m);
  const b = m.querySelector('#mbody');
  const t = m.querySelector('#mtext');
  const c = m.querySelector('#mclose');
  window.openMediaModal = (src, type, name) => {
    b.innerHTML = ''; t.textContent = name;
    if (type === 'video') {
      const v = document.createElement('video');
      v.src = src; v.controls = true; v.autoplay = true;
      v.style.maxWidth = '90vw'; v.style.maxHeight = '80vh';
      b.appendChild(v);
    } else {
      const i = document.createElement('img');
      i.src = src; i.style.maxWidth = '90vw'; i.style.maxHeight = '80vh';
      b.appendChild(i);
    }
    m.style.display = 'flex';
  };
  c.onclick = () => m.style.display = 'none';
  m.onclick = e => e.target === m && (m.style.display = 'none');
  window.__modal = true;
}
